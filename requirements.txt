import streamlit as st
import pandas as pd
import xlsxwriter
from io import BytesIO

# --- CONFIGURATION DE LA PAGE ---
st.set_page_config(page_title="SolarSizing Pro", page_icon="☀️", layout="wide")

st.title("☀️ SolarSizing Pro")
st.write("Générateur expert de tableaux de dimensionnement photovoltaïque.")

# --- BARRE LATÉRALE (PARAMÈTRES) ---
st.sidebar.header("⚙️ Paramètres du Projet")

# Liste des villes et HSP (Heures d'Ensoleillement Crête)
villes_hsp = {
    'Paris': 2.8, 'Marseille': 4.2, 'Lyon': 3.3, 'Bordeaux': 3.8,
    'Lille': 2.5, 'Strasbourg': 3.0, 'Montréal': 3.5, 'Dakar': 5.5,
    'Abidjan': 5.1, 'Tunis': 4.8, 'Alger': 4.6, 'Casablanca': 4.9,
    'Bruxelles': 2.6, 'Genève': 3.2, 'Nice': 4.5, 'Yaoundé': 4.8
}

ville = st.sidebar.selectbox("Ville d'installation", list(villes_hsp.keys()))
hsp = villes_hsp[ville]
st.sidebar.metric("HSP Moyen", f"{hsp} kWh/m²/j")

autonomie = st.sidebar.number_input("Autonomie souhaitée (Jours)", min_value=1, value=2)
tension = st.sidebar.selectbox("Tension du système (V)", [12, 24, 48])
tech_batterie = st.sidebar.selectbox("Technologie Batterie", ["Lithium (LiFePO4)", "Plomb / Gel"])
dod = 0.80 if tech_batterie == "Lithium (LiFePO4)" else 0.50
st.sidebar.write(f"Profondeur de décharge (DoD) : **{dod*100}%**")

rendement = st.sidebar.slider("Rendement global système", 0.50, 0.95, 0.75, 0.05)

# --- CORPS DE L'APPLICATION (BILAN ÉNERGÉTIQUE) ---
st.header("🔌 Bilan des Appareils")

# Données initiales pour le tableau
df_initial = pd.DataFrame({
    "Appareil": ["Éclairage LED", "Réfrigérateur", "Télévision", "Ordinateur"],
    "Quantité": [5, 1, 1, 1],
    "Puissance (W)": [10, 150, 80, 65],
    "Heures / Jour": [6, 24, 5, 8],
    "Type de charge": ["Continue", "Cyclique", "Électronique", "Électronique"]
})

# Tableau éditable
df_edited = st.data_editor(
    df_initial,
    num_rows="dynamic",
    use_container_width=True,
    column_config={
        "Quantité": st.column_config.NumberColumn(min_value=0),
        "Puissance (W)": st.column_config.NumberColumn(min_value=0),
        "Heures / Jour": st.column_config.NumberColumn(min_value=0, max_value=24),
    }
)

# Calcul de l'énergie journalière
df_edited["Énergie (Wh/j)"] = df_edited["Quantité"] * df_edited["Puissance (W)"] * df_edited["Heures / Jour"]
besoin_total = df_edited["Énergie (Wh/j)"].sum()

st.divider()

# --- CALCULS DE DIMENSIONNEMENT ---
st.header("📊 Résultats de Dimensionnement")

# 1. Panneaux Solaires
puissance_panneaux = besoin_total / (hsp * rendement)

# 2. Batteries
capacite_batteries = (besoin_total * autonomie) / (tension * dod * 0.90) # 0.90 rendement onduleur

# 3. Régulateur
courant_regulateur = (puissance_panneaux / tension) * 1.25

# 4. Onduleur
puissance_onduleur = (df_edited["Quantité"] * df_edited["Puissance (W)"]).sum() * 1.2

# Affichage des métriques
col1, col2, col3, col4 = st.columns(4)
col1.metric("Panneaux Solaires", f"{puissance_panneaux:,.0f} Wc")
col2.metric("Parc Batteries", f"{capacite_batteries:,.0f} Ah")
col3.metric("Régulateur", f"{courant_regulateur:,.0f} A")
col4.metric("Onduleur", f"{puissance_onduleur:,.0f} W")

# Alerte de sécurité
if tension == 12 and capacite_batteries > 200:
    st.error("⚠️ **ALERTE EXPERT :** Courant trop élevé en 12V. Passez sur un système 24V ou 48V pour éviter la surchauffe.")
elif tension == 24 and capacite_batteries > 400:
    st.warning("⚠️ **ALERTE EXPERT :** Courant élevé en 24V. Envisagez un passage en 48V.")
else:
    st.success("✅ Configuration de tension adaptée.")

st.divider()

# --- GÉNÉRATION DU FICHIER EXCEL ---
def create_excel_file(df, params):
    output = BytesIO()
    workbook = xlsxwriter.Workbook(output, {'in_memory': True})
    
    # Formats
    title = workbook.add_format({'bold': True, 'font_size': 16, 'align': 'center', 'valign': 'vcenter', 'bg_color': '#4472C4', 'font_color': 'white'})
    header = workbook.add_format({'bold': True, 'bg_color': '#D9E2F3', 'border': 1, 'align': 'center'})
    input_cell = workbook.add_format({'bg_color': '#FFFFCC', 'border': 1})
    result_cell = workbook.add_format({'bold': True, 'border': 1, 'num_format': '#,##0.00'})
    
    # 1. Onglet BaseVilles (Caché)
    ws_villes = workbook.add_worksheet('BaseVilles')
    ws_villes.write_column('A2', list(villes_hsp.keys()))
    ws_villes.write_column('B2', list(villes_hsp.values()))
    ws_villes.hide()
    
    # 2. Onglet Paramètres
    ws_param = workbook.add_worksheet('Parametres')
    ws_param.set_column('A:A', 35)
    ws_param.set_column('B:B', 20)
    ws_param.merge_range('A1:B1', 'PARAMETRES DU PROJET', title)
    
    param_data = [
        ("Ville d'installation", params['ville']),
        ('HSP Moyen (kWh/m²/j)', f"=RECHERCHEV(B2;BaseVilles!A:B;2;FAUX)"),
        ('Autonomie souhaitée (Jours)', params['autonomie']),
        ('Tension du système (V)', params['tension']),
        ('Technologie Batterie', params['tech_batterie']),
        ('Profondeur de décharge (DoD)', params['dod']),
        ('Rendement global système', params['rendement']),
    ]
    
    for i, (label, val) in enumerate(param_data):
        ws_param.write(i+2, 0, label, header)
        if isinstance(val, str) and val.startswith('='):
            ws_param.write_formula(i+2, 1, val)
        else:
            ws_param.write(i+2, 1, val, input_cell)
            
    ws_param.data_validation('B2', {'validate': 'list', 'source': '=BaseVilles!$A$2:$A$17'})
    ws_param.data_validation('B6', {'validate': 'list', 'source': '=["Lithium (LiFePO4)","Plomb / Gel"]'})
    
    # 3. Onglet Bilan
    ws_bilan = workbook.add_worksheet('Bilan')
    ws_bilan.set_column('A:A', 25)
    ws_bilan.set_column('B:F', 15)
    ws_bilan.merge_range('A1:F1', 'BILAN DES APPAREILS', title)
    
    headers = ['Appareil', 'Quantité', 'Puissance (W)', 'Heures/Jour', 'Énergie (Wh/j)', 'Type']
    for col, h in enumerate(headers):
        ws_bilan.write(2, col, h, header)
        
    for i, row in df.iterrows():
        ws_bilan.write(i+3, 0, row['Appareil'])
        ws_bilan.write(i+3, 1, row['Quantité'])
        ws_bilan.write(i+3, 2, row['Puissance (W)'])
        ws_bilan.write(i+3, 3, row['Heures / Jour'])
        ws_bilan.write_formula(i+3, 4, f'=B{i+4}*C{i+4}*D{i+4}')
        ws_bilan.write(i+3, 5, row.get('Type de charge', 'Standard'))
        
    ws_bilan.write('E25', 'TOTAL BESOIN JOURNALIER', header)
    ws_bilan.write_formula('F25', '=SOMME(E4:E24)', result_cell)
    
    # 4. Onglet Résultats
    ws_res = workbook.add_worksheet('Resultats')
    ws_res.set_column('A:A', 35)
    ws_res.set_column('B:B', 20)
    ws_res.set_column('C:C', 15)
    ws_res.merge_range('A1:D1', 'DIMENSIONNEMENT & RECOMMANDATIONS', title)
    
    ws_res.write('A3', '1. Générateur Photovoltaïque', header)
    ws_res.write_formula('B4', '=Bilan!F25 / (Parametres!B3 * Parametres!B8)', result_cell)
    
    ws_res.write('A6', '2. Parc Batteries', header)
    ws_res.write_formula('B7', '=(Bilan!F25 * Parametres!B4) / (Parametres!B5 * Parametres!B7)', result_cell)
    
    workbook.close()
    output.seek(0)
    return output

# Bouton de téléchargement
st.subheader("📥 Exporter le Tableau Expert")
st.write("Téléchargez le fichier Excel contenant toutes les formules dynamiques pour le modifier hors ligne.")

if st.button("Générer le fichier Excel"):
    params = {
        'ville': ville,
        'autonomie': autonomie,
        'tension': tension,
        'tech_batterie': tech_batterie,
        'dod': dod,
        'rendement': rendement
    }
    excel_file = create_excel_file(df_edited, params)
    
    st.download_button(
        label="💾 Télécharger Dimensionnement_Solaire.xlsx",
        data=excel_file,
        file_name="Dimensionnement_Solaire.xlsx",
        mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
    )
